{{- /* Previous and next page in the sidebar order (nav-order.html). The home page is not in the pager: it starts at its first child. A page outside the tree gets none. */ -}} {{- $order := partialCached "nav-order.html" site.Home -}} {{- $prev := false -}} {{- $next := false -}} {{- $found := false -}} {{- range $i, $p := $order -}} {{- if eq $p $ -}} {{- $found = true -}} {{- if gt $i 0 }}{{ $prev = index $order (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $order) }}{{ $next = index $order (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- if or $prev $next }}
{{- with $prev }} Previous← {{ .LinkTitle }} {{- end }} {{- with $next }} Next{{ .LinkTitle }} → {{- end }}
{{- end -}}
